    <script>
      /* --- 1. SETUP THREE.JS 3D BACKGROUND --- */
      const canvas = document.querySelector("#webgl-bg");
      const scene = new THREE.Scene();

      // Kamera Setup
      const camera = new THREE.PerspectiveCamera(
        75,
        window.innerWidth / window.innerHeight,
        0.1,
        1000
      );
      camera.position.z = 30;

      // Renderer
      const renderer = new THREE.WebGLRenderer({
        canvas: canvas,
        alpha: true,
        antialias: true,
      });
      renderer.setSize(window.innerWidth, window.innerHeight);
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));

      // Objek 1: Wireframe TorusKnot
      const geometry = new THREE.TorusKnotGeometry(10, 2.5, 120, 16);
      const material = new THREE.MeshStandardMaterial({
        color: 0x38bdf8,
        wireframe: true,
        transparent: true,
        opacity: 0.6,
      });
      const torusKnot = new THREE.Mesh(geometry, material);
      scene.add(torusKnot);

      // Objek 2: Partikel Bintang / Dust di Latar Belakang
      const particlesGeometry = new THREE.BufferGeometry();
      const particlesCount = 300;
      const posArray = new Float32Array(particlesCount * 3);

      for (let i = 0; i < particlesCount * 3; i++) {
        posArray[i] = (Math.random() - 0.5) * 100;
      }

      particlesGeometry.setAttribute(
        "position",
        new THREE.BufferAttribute(posArray, 3)
      );
      const particlesMaterial = new THREE.PointsMaterial({
        size: 0.15,
        color: 0x818cf8,
        transparent: true,
        opacity: 0.8,
      });
      const particlesMesh = new THREE.Points(
        particlesGeometry,
        particlesMaterial
      );
      scene.add(particlesMesh);

      // Pencahayaan
      const pointLight = new THREE.PointLight(0xffffff, 1.2);
      pointLight.position.set(20, 20, 20);
      const ambientLight = new THREE.AmbientLight(0xffffff, 0.4);
      scene.add(pointLight, ambientLight);

      // Respon Mouse Parallax
      let mouseX = 0;
      let mouseY = 0;
      window.addEventListener("mousemove", (event) => {
        mouseX = event.clientX / window.innerWidth - 0.5;
        mouseY = event.clientY / window.innerHeight - 0.5;
      });

      // Responsif saat window di-resize
      window.addEventListener("resize", () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
      });

      // Loop Render 3D
      function animate3D() {
        requestAnimationFrame(animate3D);

        // Rotasi berkesinambungan
        torusKnot.rotation.x += 0.003;
        torusKnot.rotation.y += 0.004;

        particlesMesh.rotation.y += 0.0008;

        // Efek lerp smooth mengikuti mouse
        torusKnot.rotation.x += mouseY * 0.02;
        torusKnot.rotation.y += mouseX * 0.02;

        renderer.render(scene, camera);
      }
      animate3D();

      /* --- 2. LOGIKA NAVBAR MOBILE TOGGLE --- */
      const menuToggle = document.getElementById("menu-toggle");
      const navLinksMobile = document.getElementById("nav-links");
      const menuIcon = menuToggle.querySelector("i");

      menuToggle.addEventListener("click", () => {
        navLinksMobile.classList.toggle("active");

        // Ubah ikon dari 'bars' ke 'xmark' (tanda silang) saat terbuka
        if (navLinksMobile.classList.contains("active")) {
          menuIcon.classList.remove("fa-bars");
          menuIcon.classList.add("fa-xmark");
        } else {
          menuIcon.classList.remove("fa-xmark");
          menuIcon.classList.add("fa-bars");
        }
      });

      // Otomatis menutup menu mobile saat salah satu item menu diklik
      document.querySelectorAll(".nav-links a").forEach((link) => {
        link.addEventListener("click", () => {
          navLinksMobile.classList.remove("active");
          menuIcon.classList.remove("fa-xmark");
          menuIcon.classList.add("fa-bars");
        });
      });

      /* --- 3. ANIMASI SCROLL & ACTIVE MENU OBSERVER --- */
      const fadeElements = document.querySelectorAll(".fade-in");
      const sections = document.querySelectorAll("section");
      const navLinks = document.querySelectorAll(".nav-links a");

      // Observer untuk animasi Fade-In saat scroll
      const fadeObserver = new IntersectionObserver(
        (entries) => {
          entries.forEach((entry) => {
            if (entry.isIntersecting) {
              entry.target.classList.add("visible");
            }
          });
        },
        { threshold: 0.15 }
      );

      fadeElements.forEach((el) => fadeObserver.observe(el));

      // Highlight menu aktif sesuai posisi scroll
      window.addEventListener("scroll", () => {
        let current = "";
        sections.forEach((section) => {
          const sectionTop = section.offsetTop;
          const sectionHeight = section.clientHeight;
          if (pageYOffset >= sectionTop - sectionHeight / 3) {
            current = section.getAttribute("id");
          }
        });

        navLinks.forEach((link) => {
          link.classList.remove("active");
          if (link.getAttribute("href") === `#${current}`) {
            link.classList.add("active");
          }
        });
      });

<script>
  /* --- FETCH RESUME & PROJECTS OTOMATIS DAAR GOOGLE SHEETS --- */
const API_RESUME = "https://api.sheetbest.com/sheets/677180fe-0dac-46e1-8da4-fcbd00ae05b7";
const API_PROJECTS = "PASANG_URL_SHEETBEST_TAB_PROJECTS";

async function loadDynamicContent() {
    // 1. Render Resume
    try {
        const resResume = await fetch(API_RESUME);
        const dataResume = await resResume.json();
        const resumeContainer = document.getElementById('sheet-resume-container');
        
        if (dataResume.length > 0) {
            resumeContainer.innerHTML = '';
            dataResume.sort((a,b) => a.urutan - b.urutan).forEach(item => {
                resumeContainer.innerHTML += `
                    <div class="timeline-item fade-in visible">
                        <div class="timeline-card">
                            <span class="timeline-date">${item.periode}</span>
                            <h4>${item.posisi}</h4>
                            <div class="timeline-company">${item.perusahaan}</div>
                            <p>${item.deskripsi}</p>
                        </div>
                    </div>`;
            });
        }
    } catch (err) { console.error("Error Resume:", err); }

    // 2. Render Projects
    try {
        const resProjects = await fetch(API_PROJECTS);
        const dataProjects = await resProjects.json();
        const projectsContainer = document.querySelector('.projects-grid');

        if (dataProjects.length > 0) {
            projectsContainer.innerHTML = '';
            dataProjects.sort((a,b) => a.urutan - b.urutan).forEach(item => {
                const tagsHTML = item.tags ? item.tags.split(',').map(tag => `<span class="tag">${tag.trim()}</span>`).join('') : '';
                
                projectsContainer.innerHTML += `
                    <div class="project-card fade-in visible">
                        <div class="project-img-wrapper">
                            <svg viewBox="0 0 400 200" width="100%" height="100%">
                                <rect width="400" height="200" fill="#0f172a"/>
                                <circle cx="200" cy="100" r="50" fill="none" stroke="#38bdf8" stroke-width="2"/>
                            </svg>
                            <span class="project-badge">${item.kategori}</span>
                        </div>
                        <div class="project-info">
                            <h3>${item.judul}</h3>
                            <p>${item.deskripsi}</p>
                            <div class="tech-tags">${tagsHTML}</div>
                            <div class="project-links">
                                <a href="${item.link_demo || '#'}" target="_blank" class="project-btn btn-demo"><i class="fa-solid fa-arrow-up-right-from-square"></i> Demo</a>
                                <a href="${item.link_github || '#'}" target="_blank" class="project-btn btn-code"><i class="fa-brands fa-github"></i> Code</a>
                            </div>
                        </div>
                    </div>`;
            });
        }
    } catch (err) { console.error("Error Projects:", err); }
}
